W tej wersji bota użytkownicy mogą zarządzać rangami w jednostkach za pomocą komend -pm (promocja) i -dm (degradacja). Bot przypisuje odpowiednie role do użytkowników, zaczynając od najniższej rangi w jednostce (np. "Private"), a następnie aktualizuje ich nicki, dodając odpowiednie przedrostki (np. "B-7"). Jeśli użytkownik nie ma roli w jednostce, bot przypisuje mu "Private" i odpowiedni przedrostek. Bot usuwa przedrostki, gdy użytkownik jest zdegradowany lub usunięty z jednostki. Wszystkie zmiany są komunikowane administratorowi, a komendy są dostępne tylko dla uprawnionych użytkowników.

# Funkcja do logowania do pliku
def log_to_file(message):
    with open("log.txt", "a", encoding="utf-8") as log_file:
        log_file.write(f"{datetime.datetime.now()} - {message}\n")

def generate_secret_code():
    return ''.join(random.choices(string.ascii_letters + string.digits, k=12))

# Hierarchia ról administracyjnych (od najniższej do najwyższej)
admin_role_hierarchy = [
    'MŁODSZY ADMIN',
    'ADMIN',
    'HEAD ADMIN',
    'CO-OWNER',
    'OWNER'
]

def get_highest_admin_role(member):
    """Zwraca najwyższą rolę administracyjną użytkownika."""
    for role_name in reversed(admin_role_hierarchy):  # Od najwyższej do najniższej
        role = discord.utils.get(member.roles, name=role_name)
        if role:
            return role.name
    return None

def get_next_admin_rank(current_rank):
    try:
        index = admin_role_hierarchy.index(current_rank)
        if index + 1 < len(admin_role_hierarchy):
            return admin_role_hierarchy[index + 1]
    except ValueError:
        return None
    return None

def get_previous_admin_rank(current_rank):
    try:
        index = admin_role_hierarchy.index(current_rank)
        if index - 1 >= 0:
            return admin_role_hierarchy[index - 1]
    except ValueError:
        return None
    return None

# Komenda adminpromote
@bot.command()
async def adminpromote(ctx, member: discord.Member):
    author_highest_role = get_highest_admin_role(ctx.author)
    if not author_highest_role:
        await ctx.send("❌ Nie masz wystarczających uprawnień.")
        return

    member_highest_role = get_highest_admin_role(member)
    if not member_highest_role:
        current_role = 'MŁODSZY ADMIN'
    else:
        current_role = member_highest_role

    next_role = get_next_admin_rank(current_role)
    if not next_role:
        await ctx.send(f"⚠️ {member.mention} ma już najwyższą możliwą rangę administracyjną.")
        return

    # POPRAWNA KONTROLA DOSTĘPU:
    if admin_role_hierarchy.index(author_highest_role) <= admin_role_hierarchy.index(next_role):
        await ctx.send(f"❌ Nie możesz awansować użytkownika do roli **{next_role}**, ponieważ jest ona taka sama lub wyższa od twojej.")
        return

    role_current = discord.utils.get(ctx.guild.roles, name=current_role)
    role_next = discord.utils.get(ctx.guild.roles, name=next_role)

    if role_next:
        if role_current:
            await member.remove_roles(role_current)
        await member.add_roles(role_next)
        await ctx.send(f"✅ Awansowano {member.mention} na **{next_role}**.")
    else:
        await ctx.send("❌ Wystąpił błąd przy zmianie roli.")

# Komenda -getroles / alias -r
@bot.command(aliases=['r'])
async def getroles(ctx, user: discord.Member):
    log_to_file(f"[GETROLES] Komenda użyta na kanale: {ctx.channel} przez {ctx.author}")

    # Lista wszystkich ról z role_hierarchy
    allowed_roles = []
    for ranks in role_hierarchy.values():
        allowed_roles.extend(ranks)

    # Filtrujemy role użytkownika, które są w allowed_roles
    user_roles = [role.name for role in user.roles if role.name in allowed_roles]

    if not user_roles:
        await ctx.send(f"⚠️ {user.mention} nie ma żadnych rang jednostek.")
    else:
        embed = discord.Embed(
            title=f"Role jednostek użytkownika {user.name}",
            description="\n".join(f"- {r}" for r in user_roles),
            color=discord.Color.blue()
        )
        embed.set_footer(text=f"Komenda użyta na kanale: {ctx.channel}")
        await ctx.send(embed=embed)

# Funkcje pomocnicze do awansów i degradacji
def get_next_rank(unit, current_rank):
    try:
        unit_roles = role_hierarchy[unit]
        current_index = unit_roles.index(current_rank.strip())
        if current_index + 1 < len(unit_roles):
            return unit_roles[current_index + 1]
    except ValueError:
        print(f"Nie znaleziono '{current_rank}' w {unit_roles}")
        return None
    return None

def get_previous_rank(unit, current_rank):
    try:
        unit_roles = role_hierarchy[unit]
        current_index = unit_roles.index(current_rank.strip())
        if current_index - 1 >= 0:
            return unit_roles[current_index - 1]
    except ValueError:
        print(f"Nie znaleziono '{current_rank}' w {unit_roles}")
        return None
    return None

def has_promotion_permission(member, unit):
    allowed_ranks = ["Head Captain", "Captain", "Sergeant Major", "Sergeant"]

    unit_fullname = unit_fullnames[unit.lower()]

    for role in member.roles:
        if unit_fullname in role.name:
            for rank in allowed_ranks:
                if role.name.startswith(f"{unit_fullname} - {rank}"):
                    return True
    return False

def get_unit_key_from_prefix(prefix):
    for key, greek in greeks.items():
        if greek == prefix:
            return key
    if prefix == "RRT":
        return "rrt"
    return prefix  # fallback

def add_unit_prefix_to_nick(user, unit_prefix):
    current_nick = user.nick if user.nick else user.name
    parts = current_nick.split()

    # Aktualne prefixy
    current_prefixes = [p for p in parts if p in greeks.values() or p == "RRT"]

    if unit_prefix in current_prefixes:
        return current_nick

    all_prefixes = list(set(current_prefixes + [unit_prefix]))
    all_prefixes_sorted = sorted(all_prefixes, key=lambda p: unit_hierarchy.get(get_unit_key_from_prefix(p), 999))

    # Pozostała część nicku
    rest = " ".join(part for part in parts if part not in current_prefixes)

    new_nick = " ".join(all_prefixes_sorted) + " " + rest
    return new_nick.strip()
    
def remove_unit_prefix_from_nick(user, unit_prefix):
    current_nick = user.nick if user.nick else user.name
    parts = current_nick.split()

    # Usuń prefix
    parts = [p for p in parts if p != unit_prefix]

    return " ".join(parts).strip()    

# Komenda -promote / alias -pm
@bot.command(aliases=['pm'])
async def promote(ctx, user: discord.Member, unit: str):
    # Zamiana jednostki na małe litery
    unit = unit.lower()

    # Sprawdzanie, czy jednostka istnieje
    if unit not in unit_fullnames:
        await ctx.send("❌ Nieprawidłowa jednostka.")
        return

    # Sprawdzanie uprawnień autora komendy
    if not has_promotion_permission(ctx.author, unit):
        await ctx.send("❌ Nie masz uprawnień, aby awansować w tej jednostce!")
        return

    unit_fullname = unit_fullnames[unit]

    # Sprawdzanie czy użytkownik ma już rolę w tej jednostce
    current_role = None
    for role in user.roles:
        if role.name.startswith(unit_fullname):
            current_role = role.name
            break

    if not current_role:
        # Użytkownik nie ma żadnej roli -> Dodajemy "Recruit"
        role_to_add = discord.utils.get(ctx.guild.roles, name=f"{unit_fullname} - Private")
        if role_to_add:
            await user.add_roles(role_to_add)
            await user.edit(nick=add_unit_prefix_to_nick(user, unit_fullname))
            await ctx.send(f"✅ {user.mention} został dodany do jednostki {unit_fullname} jako **Private**.")
        else:
            await ctx.send(f"❌ Nie znaleziono roli {unit_fullname} - Private.")
        return

    # Ma już rolę, próbujemy awansować
    next_rank = get_next_rank(unit, current_role)
    if not next_rank:
        await ctx.send(f"⚠️ {user.mention} ma już najwyższą rangę w {unit_fullname}.")
        return

    role_current = discord.utils.get(ctx.guild.roles, name=current_role)
    role_next = discord.utils.get(ctx.guild.roles, name=next_rank)

    if role_current and role_next:
        await user.remove_roles(role_current)
        await user.add_roles(role_next)
        await ctx.send(f"✅ Awansowano {user.mention} na **{role_next.name}**.")
    else:
        await ctx.send("❌ Wystąpił błąd przy zmianie ról.")


# Komenda -demote / alias -dm
@bot.command(aliases=['dm'])
async def demote(ctx, user: discord.Member, unit: str):
    # Zamiana jednostki na małe litery
    unit = unit.lower()

    # Sprawdzanie, czy jednostka istnieje
    if unit not in unit_fullnames:
        await ctx.send("❌ Nieprawidłowa jednostka.")
        return

    # Sprawdzanie uprawnień autora komendy
    if not has_promotion_permission(ctx.author, unit):
        await ctx.send("❌ Nie masz uprawnień, aby degradować w tej jednostce!")
        return

    unit_fullname = unit_fullnames[unit]

    # Szukanie obecnej roli użytkownika
    current_role = None
    for role in user.roles:
        if role.name.startswith(unit_fullname):
            current_role = role.name
            break

    if not current_role:
        await ctx.send(f"⚠️ {user.mention} nie ma przypisanej rangi w {unit_fullname}.")
        return

    if current_role == f"{unit_fullname} - Private":
        # Jeśli jest już na najniższej randze -> Usuwamy rolę i prefix
        role_to_remove = discord.utils.get(ctx.guild.roles, name=current_role)
        if role_to_remove:
            await user.remove_roles(role_to_remove)
            await user.edit(nick=remove_unit_prefix_from_nick(user, unit_fullname))
            await ctx.send(f"⚠️ {user.mention} był na najniższej randze. Usunięto go z jednostki {unit_fullname}.")
        return

    previous_rank = get_previous_rank(unit, current_role)
    if not previous_rank:
        await ctx.send(f"⚠️ {user.mention} już jest na najniższej randze.")
        return

    role_current = discord.utils.get(ctx.guild.roles, name=current_role)
    role_previous = discord.utils.get(ctx.guild.roles, name=previous_rank)

    if role_current and role_previous:
        await user.remove_roles(role_current)
        await user.add_roles(role_previous)
        await ctx.send(f"✅ Zdegradowano {user.mention} na **{role_previous.name}**.")
    else:
        await ctx.send("❌ Wystąpił błąd przy zmianie ról.")
